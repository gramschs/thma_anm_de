import numpy as np

# f(x) = x^2: f(1.5) = 2.25, f(2.0) = 4.0, f(2.5) = 6.25
# Vorwärtsdifferenz: (6.25 - 4.0) / 0.5 = 4.5,  Fehler = |4 - 4.5| = 0.5

    """Analytische Ableitung von x^2: 2x."""

Zu Frage 1: Der zentrale Differenzenquotient liefert für f(x)=x2f(x) = x^2f(x)=x2 das exakte Ergebnis. Das liegt daran, dass f(x)=x2f(x) = x^2f(x)=x2 ein Polynom zweiten Grades ist: sein führender Fehlerterm enthält f′′′(x)f'''(x)f′′′(x), und für ein Polynom zweiten Grades gilt f′′′=0f''' = 0f′′′=0. Der Fehler ist damit exakt null, unabhängig von Δx\Delta xΔx.

Zu Frage 3: Der Fehler der zentralen Formel ist für alle drei Schrittweiten numerisch null (bis auf Maschinengenauigkeit). Vorwärts- und Rückwärtsdifferenz haben dagegen einen Fehler proportional zu Δx\Delta xΔx.

Übung 7.2 (✩✩)

Gegeben ist die Funktion f(x)=ln⁡(x)f(x) = \ln(x)f(x)=ln(x) mit der analytischen Ableitung f′(x)=1/xf'(x) = 1/xf′(x)=1/x. An der Stelle x0=2x_0 = 2x0​=2 gilt also f′(2)=0.5f'(2) = 0.5f′(2)=0.5.

  2. Berechnen Sie für die Vorwärtsdifferenz den theoretisch vorhergesagten Fehler laut Taylor: Δx2⋅∣f′′(x0)∣\frac{\Delta x}{2} \cdot |f''(x_0)|2Δx​⋅∣f′′(x0​)∣. Die zweite Ableitung von ln⁡(x)\ln(x)ln(x) ist f′′(x)=−1/x2f''(x) = -1/x^2f′′(x)=−1/x2, also f′′(2)=−0.25f''(2) = -0.25f′′(2)=−0.25. Vergleichen Sie den theoretischen mit dem tatsächlichen Fehler.

# Code-Zelle
Lösung
import numpy as np
import matplotlib.pyplot as plt

# --- Funktionen definieren ---
def f(x):
    return np.log(x)

def f_strich(x):
    """Analytische Ableitung von ln(x): 1/x."""
    return 1 / x

def f_zweimal(x):
    """Zweite Ableitung von ln(x): -1/x^2."""
    return -1 / x**2

def vorwaerts(f, x, dx):
    return (f(x + dx) - f(x)) / dx

def rueckwaerts(f, x, dx):
    return (f(x) - f(x - dx)) / dx

def zentral(f, x, dx):
    return (f(x + dx) - f(x - dx)) / (2 * dx)

# Auswertestelle und exakter Wert
x0    = 2.0
exakt = f_strich(x0)   # = 0.5

# --- Frage 1: Fehlertabelle ---
print("Fehler der drei Differenzenformeln für ln(x) bei x0 = 2:")
for dx in [0.1, 0.01, 0.001, 0.0001]:
    fv = abs(exakt - vorwaerts(f, x0, dx))
    fr = abs(exakt - rueckwaerts(f, x0, dx))
    fz = abs(exakt - zentral(f, x0, dx))
    print(f"  dx = {dx:.4f}: Vorwärts = {fv:.2e}, "
          f"Rückwärts = {fr:.2e}, Zentral = {fz:.2e}")

print()

# --- Frage 2: Taylor-Vorhersage für Vorwärtsdifferenz ---
print("Vergleich: tatsächlicher Fehler vs. Taylor-Vorhersage (Vorwärts):")
for dx in [0.1, 0.01, 0.001, 0.0001]:
    # Tatsächlicher Fehler
    fehler_num  = abs(exakt - vorwaerts(f, x0, dx))
    # Taylor-Vorhersage: (dx/2) * |f''(x0)|
    fehler_theo = (dx / 2) * abs(f_zweimal(x0))
    print(f"  dx = {dx:.4f}: tatsächlich = {fehler_num:.2e}, "
          f"Theorie = {fehler_theo:.2e}")

# --- Frage 3: Log-Log-Plot ---
dx_werte = np.logspace(-1, -8, 100)

fehler_v = []
fehler_r = []
fehler_z = []
for dx in dx_werte:
    fehler_v.append(abs(exakt - vorwaerts(f, x0, dx)))
    fehler_r.append(abs(exakt - rueckwaerts(f, x0, dx)))
    fehler_z.append(abs(exakt - zentral(f, x0, dx)))

fig, ax = plt.subplots()
ax.loglog(dx_werte, fehler_v, label='Vorwärtsdifferenz')
ax.loglog(dx_werte, fehler_r, label='Rückwärtsdifferenz')
ax.loglog(dx_werte, fehler_z, label='Zentraler Differenzenquot.')
ax.set_xlabel('Schrittweite Δx')
ax.set_ylabel('Fehler |f′(x₀) − Näherung|')
ax.set_title('Fehlerordnungen für f(x) = ln(x)')
ax.legend()
ax.grid(True, which='both')
plt.show()

Im Log-Log-Plot haben Vorwärts- und Rückwärtsdifferenz eine Steigung von etwa 1: ihr Fehler ist proportional zu Δx\Delta xΔx (Fehlerordnung O(Δx)O(\Delta x)O(Δx)). Der zentrale Differenzenquotient hat eine Steigung von etwa 2: sein Fehler ist proportional zu Δx2\Delta x^2Δx2 (Fehlerordnung O(Δx2)O(\Delta x^2)O(Δx2)). Die Taylor-Vorhersage für die Vorwärtsdifferenz stimmt gut mit dem tatsächlichen Fehler überein.

    • An den inneren Punkten (Index 1 bis 5): zentralen Differenzenquotienten.

  3. Stellen Sie Temperatur und Temperaturgradient in zwei übereinander angeordneten Teilplots (fig, (ax1, ax2) = plt.subplots(2, 1)) dar.

# Code-Zelle
Lösung
import numpy as np
import matplotlib.pyplot as plt

# --- Eingabe: Messdaten ---
x_pos  = np.array([0.0, 5.0, 10.0, 15.0, 20.0, 25.0, 30.0])  # Positionen in cm
T_mess = np.array([85.0, 73.0, 64.0, 57.0, 52.0, 48.0, 45.0]) # Temperaturen in °C
dx     = 5.0   # Abstand zwischen den Messpunkten in cm
n      = len(T_mess)

# --- Verarbeitung: Temperaturgradient ---
# Wir legen ein leeres Array für den Gradienten an
gradient = np.zeros(n)

# Randpunkte: einseitige Differenzen
gradient[0]  = (T_mess[1] - T_mess[0]) / dx       # Vorwärtsdifferenz am linken Rand
gradient[-1] = (T_mess[-1] - T_mess[-2]) / dx     # Rückwärtsdifferenz am rechten Rand

# Innere Punkte: zentraler Differenzenquotient
for i in range(1, n - 1):
    gradient[i] = (T_mess[i + 1] - T_mess[i - 1]) / (2 * dx)

# Position des größten Gradienten (dem Betrag nach)
idx_max = np.argmax(np.abs(gradient))
print(f"Größter Gradient bei x = {x_pos[idx_max]:.0f} cm: "
      f"{gradient[idx_max]:.2f} °C/cm")

# --- Ausgabe: zwei Teilplots übereinander ---
fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(6, 6))

# Oberer Plot: Temperaturprofil
ax1.plot(x_pos, T_mess, marker='o', color='tab:red')
ax1.set_xlabel('Position in cm')
ax1.set_ylabel('Temperatur in °C')
ax1.set_title('Temperaturprofil des Kühlkörpers')
ax1.grid(True)

# Unterer Plot: Temperaturgradient
ax2.plot(x_pos, gradient, marker='o', color='tab:blue')
ax2.set_xlabel('Position in cm')
ax2.set_ylabel('Gradient in °C/cm')
ax2.set_title('Temperaturgradient')
ax2.grid(True)

# Abstand zwischen den Plots anpassen, damit die Beschriftungen nicht überlappen
fig.tight_layout()
plt.show()

Der Temperaturgradient ist am linken Rand (x=0x = 0x=0 cm) dem Betrag nach am größten, weil dort der Kühlkörper am heißesten ist und die Temperatur am stärksten abfällt. Zum rechten Rand hin flacht das Temperaturprofil ab, der Gradient wird kleiner.

import numpy as np
import matplotlib.pyplot as plt

                   42.87, 43.83, 44.49, 44.87, 44.97])

n = len(s_mess)

# --- Verarbeitung: Geschwindigkeit ---
# Ableitung der Position nach der Zeit: v = ds/dt
v = np.zeros(n)

# Randpunkte: einseitige Differenzen
v[0]  = (s_mess[1] - s_mess[0]) / dt        # Vorwärtsdifferenz
v[-1] = (s_mess[-1] - s_mess[-2]) / dt      # Rückwärtsdifferenz

for i in range(1, n - 1):

# Anfangsgeschwindigkeit in km/h ausgeben (Umrechnung: 1 m/s = 3.6 km/h)
print(f"Anfangsgeschwindigkeit: {v[0]:.1f} m/s = {v[0] * 3.6:.1f} km/h")

# --- Verarbeitung: Beschleunigung ---

a = np.zeros(n)

# Mittlere Verzögerung (negativer Wert = Bremsen)

ax2.plot(t, v, marker='o', color='tab:orange')
