# 4.4 Ein Fachwerk mit fünf Knoten

In Kapitel 4.3 haben wir das FEM-Schema an einem Dreiecksfachwerk eingeführt.
Das Vorgehen lässt sich ohne jede Änderung am Algorithmus auf beliebig komplexe
Strukturen übertragen. Die Größe des Systems wächst mit der Anzahl der Knoten,
aber `np.linalg.solve` übernimmt den rechnerischen Aufwand.

In diesem Kapitel analysieren wir ein Fachwerk mit fünf Knoten und sechs
Stäben, wie es in vereinfachter Form in Dachkonstruktionen oder Brückenträgern
vorkommt. Neu ist die Konnektivitätsmatrix für eine komplexere Topologie und
die Frage, welche Stäbe besonders stark belastet werden.

## Setup: Geometrie, Lasten und Verbindungen

Wir betrachten fünf Knoten, von denen Knoten 0 und 3 fest gelagert sind.
An Knoten 2 greift eine Vertikalkraft von 3000 N nach unten an.

```{code-cell} python
import numpy as np
import matplotlib.pyplot as plt

# --- Materialeigenschaften ---
E = 2.1e11      # Elastizitätsmodul in N/m²
d = 1.0e-2      # Stabdurchmesser in m
A = np.pi * 0.25 * d**2

# --- Knotenkoordinaten ---
# Knoten 0: (0,0)  Knoten 1: (1,0)  Knoten 2: (2,0)
# Knoten 3: (0,1)  Knoten 4: (1,1)
NodePos = np.array([
    [0.,  1.,  2.,  0.,  1.],   # x
    [0.,  0.,  0.,  1.,  1.],   # y
])
numNodes = NodePos.shape[1]

# --- Lagerknoten ---
FixedNodeIndices = [0, 3]

# --- Konnektivitätsmatrix ---
# Jede 1 bedeutet einen Stab zwischen den entsprechenden Knoten.
# Die Topologie ergibt 6 Stäbe:
#   0-1 (unten links), 1-2 (unten rechts), 3-4 (oben),
#   1-3 (diagonal), 1-4 (vertikal), 2-4 (diagonal)
C = np.zeros((numNodes, numNodes))
C[0, 1] = 1
C[1, 2] = 1
C[1, 3] = 1
C[1, 4] = 1
C[2, 4] = 1
C[3, 4] = 1
C = C + C.T

# --- Kraftvektor ---
extForcesOnNodes = np.zeros((numNodes, 2))
extForcesOnNodes[2, 1] = -3000.     # 3000 N nach unten an Knoten 2
FTruss = extForcesOnNodes.flatten()

print("Konnektivitätsmatrix C:")
print(C.astype(int))
print(f"\nAnzahl Stäbe: {int(C.sum() / 2)}")


# --- Hilfsfunktion: Fachwerk visualisieren ---
def plotTruss(UTruss=None, scalingFactor=500, displayForces=False,
              FTruss_plot=None, title=''):
    if UTruss is None:
        UTruss = np.zeros(2 * numNodes)
    if FTruss_plot is None:
        FTruss_plot = np.zeros(2 * numNodes)

    fig, ax = plt.subplots(figsize=(8, 5))
    NodePos_d = NodePos + scalingFactor * UTruss.reshape((numNodes, 2)).T

    for i in range(numNodes):
        for j in range(i + 1, numNodes):
            if C[i, j]:
                ax.plot([NodePos[0,i],   NodePos[0,j]],
                        [NodePos[1,i],   NodePos[1,j]],
                        color='gray', linewidth=1.5, alpha=0.3)
                ax.plot([NodePos_d[0,i], NodePos_d[0,j]],
                        [NodePos_d[1,i], NodePos_d[1,j]],
                        color='tab:blue', linewidth=2.5)

    ax.scatter(NodePos[0,:],   NodePos[1,:],   c='gray',    s=60, zorder=4, alpha=0.3)
    ax.scatter(NodePos_d[0,:], NodePos_d[1,:], c='tab:red', s=80, zorder=5)
    for n in range(numNodes):
        ax.text(NodePos_d[0,n] + 0.03, NodePos_d[1,n] + 0.03, f'K{n}', fontsize=9)

    h, w = 0.10, 0.10
    for n in FixedNodeIndices:
        tx = [NodePos_d[0,n], NodePos_d[0,n]-w/2, NodePos_d[0,n]+w/2]
        ty = [NodePos_d[1,n], NodePos_d[1,n]-h,   NodePos_d[1,n]-h]
        ax.fill(tx, ty, color='tab:green', alpha=0.7)

    if displayForces:
        for n in range(numNodes):
            ax.text(NodePos_d[0,n] + 0.03, NodePos_d[1,n] - 0.10,
                    f'Fy={FTruss_plot[2*n+1]:.0f} N', fontsize=7)

    ax.set_title(title)
    ax.set_aspect('equal')
    ax.grid(True)
    plt.tight_layout()
    plt.show()


plotTruss(title='Fachwerk in Ausgangslage')
```

## Steifigkeitsmatrix, Lösung und Stabkräfte

Der Aufbau von $\mathbf{K}_\text{Truss}$, die Reduktion und das Lösen
laufen identisch zu Kapitel 4.3 ab. Wir erweitern die Auswertung um eine
neue Größe: die **Stabkraft** $N_s$ in jedem Stab.

```{code-cell} python
# --- Globale Steifigkeitsmatrix ---
KTruss = np.zeros((2 * numNodes, 2 * numNodes))

for i in range(numNodes):
    for j in range(i + 1, numNodes):
        if C[i, j]:
            diff   = NodePos[:, j] - NodePos[:, i]
            laenge = np.linalg.norm(diff)
            phi    = np.arctan2(diff[1], diff[0])
            k      = E * A / laenge
            cp, sp = np.cos(phi), np.sin(phi)
            Kij    = k * np.array([[cp**2, sp*cp], [sp*cp, sp**2]])

            KTruss[2*i:2*(i+1), 2*i:2*(i+1)] += Kij
            KTruss[2*j:2*(j+1), 2*j:2*(j+1)] += Kij
            KTruss[2*i:2*(i+1), 2*j:2*(j+1)] -= Kij
            KTruss[2*j:2*(j+1), 2*i:2*(i+1)] -= Kij

# --- Reduziertes LGS lösen ---
FreeNodeIndices = [n for n in range(numNodes) if n not in FixedNodeIndices]
reduceMask = []
for n in FreeNodeIndices:
    reduceMask += [2*n, 2*n + 1]
reduceMask = np.array(reduceMask)

Ured = np.linalg.solve(KTruss[reduceMask, :][:, reduceMask], FTruss[reduceMask])

UTruss = np.zeros(2 * numNodes)
UTruss[reduceMask] = Ured

FTruss_result = np.round(KTruss @ UTruss, 6)

# --- Stabkräfte berechnen ---
# Die Stabkraft ist die axiale Kraft im Stab: N = k * Dehnung
# Dehnung = Projektion der relativen Knotenverschiebung auf die Stabachse
print("Stabkräfte (positiv = Zug, negativ = Druck):")
print(f"{'Stab':>6}  {'Länge':>8}  {'N in N':>12}  {'Typ':>8}")
print("-" * 42)

for i in range(numNodes):
    for j in range(i + 1, numNodes):
        if C[i, j]:
            diff   = NodePos[:, j] - NodePos[:, i]
            laenge = np.linalg.norm(diff)
            phi    = np.arctan2(diff[1], diff[0])
            k      = E * A / laenge

            # Relative Verschiebung des Stabs (Endknoten minus Anfangsknoten)
            ui = UTruss[2*i : 2*(i+1)]
            uj = UTruss[2*j : 2*(j+1)]
            e  = np.array([np.cos(phi), np.sin(phi)])   # Einheitsvektor entlang Stab

            # Dehnung = Projektion der relativen Verschiebung auf Stabachse
            delta = np.dot(e, uj - ui)
            N     = k * delta    # Stabkraft

            stabtyp = 'Zug' if N > 0 else 'Druck'
            print(f"{i}-{j:>1}   {laenge:>8.3f} m  {N:>12.2f} N  {stabtyp:>8}")

# --- Verformtes Fachwerk ---
plotTruss(UTruss=UTruss, scalingFactor=500,
          displayForces=True, FTruss_plot=FTruss_result,
          title='Verformtes Fachwerk (Skalierungsfaktor = 500)')
```

Eine positive Stabkraft bedeutet **Zug** (der Stab wird gestreckt), eine
negative bedeutet **Druck** (der Stab wird gestaucht). Gedrückte Stäbe
müssen auf Knicken ausgelegt werden; das ist in der Strukturmechanik
ein eigenständiges Dimensionierungsproblem.

````{admonition} Lösung
```python
import numpy as np

# Frage 1: Stabkräfte wurden bereits ausgegeben.
# Stäbe auf der belasteten Seite (rechts) erwarten wir unter Zug oder
# Druck, abhängig von der Geometrie. Stab 1-2 (horizontal unten rechts)
# steht typischerweise unter Druck, Stäbe 2-4 und 1-4 unter Zug/Druck
# entsprechend der Kraftübertragungslinie.

# Frage 2: doppelter Durchmesser -> A_neu = 4 * A (Fläche ~ d²)
# k_neu = E * A_neu / l = 4 * k -> U_neu = F / (4k) = U / 4
d_neu = 2.e-2
A_neu = np.pi * 0.25 * d_neu**2

# KTruss neu aufbauen mit A_neu
KTruss_neu = np.zeros((2 * numNodes, 2 * numNodes))
for i in range(numNodes):
    for j in range(i + 1, numNodes):
        if C[i, j]:
            diff   = NodePos[:, j] - NodePos[:, i]
            laenge = np.linalg.norm(diff)
            phi    = np.arctan2(diff[1], diff[0])
            k_neu  = E * A_neu / laenge
            cp, sp = np.cos(phi), np.sin(phi)
            Kij    = k_neu * np.array([[cp**2, sp*cp],[sp*cp, sp**2]])
            KTruss_neu[2*i:2*(i+1), 2*i:2*(i+1)] += Kij
            KTruss_neu[2*j:2*(j+1), 2*j:2*(j+1)] += Kij
            KTruss_neu[2*i:2*(i+1), 2*j:2*(j+1)] -= Kij
            KTruss_neu[2*j:2*(j+1), 2*i:2*(i+1)] -= Kij

Ured_neu  = np.linalg.solve(KTruss_neu[reduceMask, :][:, reduceMask],
                             FTruss[reduceMask])
UTruss_neu = np.zeros(2 * numNodes)
UTruss_neu[reduceMask] = Ured_neu

print(f"u2y (d=1cm): {UTruss[5]*1e3:.4f} mm")
print(f"u2y (d=2cm): {UTruss_neu[5]*1e3:.4f} mm")
print(f"Faktor:      {UTruss[5] / UTruss_neu[5]:.1f}")
# -> Faktor 4, weil A ~ d² und k ~ A, also U ~ 1/k ~ 1/d²
```

Doppelter Durchmesser bedeutet vierfache Querschnittsfläche ($A \propto d^2$),
also vierfache Stabsteifigkeit $k = EA/l$. Da die Verschiebung umgekehrt
proportional zur Steifigkeit ist ($u \propto 1/k$), reduziert sich die
Absenkung auf ein Viertel.
````

## Zusammenfassung und Ausblick

Das FEM-Schema für ebene Fachwerke ist vollständig skalierbar: egal ob 3
oder 30 Knoten, der Algorithmus bleibt identisch. Neu in diesem Kapitel
war die Berechnung der Stabkräfte aus den Knotenverschiebungen und die
Unterscheidung zwischen Zug und Druck. Die Stabkraft $N = k \cdot \delta$
folgt direkt aus der projizierten Relativverschiebung der Endknoten.

Im Übungskapitel wird das Gelernte auf eine vollständige Fachwerkbrücke
angewendet, bei der zusätzlich das Eigengewicht der Stäbe als verteilte
Last berücksichtigt werden muss.
