# 4.5 Übung: Eine Fachwerkbrücke mit Eigengewicht

In den Kapiteln 4.3 und 4.4 haben wir Fachwerke analysiert, bei denen das
Eigengewicht der Stäbe vernachlässigt wurde. Bei realen Brücken ist das
Eigengewicht der Konstruktion oft größer als die Nutzlast. Wir müssen es
daher in den Kraftvektor einbeziehen.

Das Grundprinzip der Finite-Elemente-Methode bleibt unverändert. Neu ist
nur, wie wir die verteilte Gewichtslast der Stäbe auf die Knoten übertragen.

## Brückengeometrie und Parameter

Wir analysieren eine Fachwerkbrücke mit zehn Knoten. Die geraden Knoten (2,
4, 6, 8) bilden den Obergurt, die ungeraden Knoten (1, 3, 5, 7) den
Untergurt. Die Knoten 0 und 9 sind die Lager.

```{code-cell} python
import numpy as np
import matplotlib.pyplot as plt

# --- Materialeigenschaften und Geometrieparameter ---
E   = 2.1e11    # Elastizitätsmodul Stahl in N/m²
d   = 1.0e-2    # Stabdurchmesser in m
rho = 7800.     # Dichte Stahl in kg/m³
g   = 9.81      # Erdbeschleunigung in m/s²
a   = 1.0       # horizontaler Abstand zwischen Knoten in m
h   = 0.7       # Höhe des Obergurts in m

A    = np.pi * 0.25 * d**2          # Querschnittsfläche in m²
diag = np.sqrt(a**2 + h**2)         # Länge der Diagonalstäbe in m

n = 10  # Anzahl der Knoten

# --- Knotenkoordinaten ---
# Knoten 0: linkes Lager  (0, 0)
# Knoten 1: (1.0, 0)  | Knoten 2: (1.0, h)   <- Paar an x=1
# Knoten 3: (2.0, 0)  | Knoten 4: (2.0, h)   <- Paar an x=2
# Knoten 5: (3.0, 0)  | Knoten 6: (3.0, h)   <- Paar an x=3
# Knoten 7: (4.0, 0)  | Knoten 8: (4.0, h)   <- Paar an x=4
# Knoten 9: rechtes Lager (5.0, 0)
NodePos = np.zeros((2, n))
for i in range(n):
    if i % 2 == 0 and i != 0:           # gerader Index (ausser 0): Obergurt
        NodePos[1, i] = h
        NodePos[0, i] = i * a / 2
    else:                                # ungerader Index + Knoten 0: Untergurt
        NodePos[0, i] = (i + 1) * a / 2
NodePos[0, 0] = 0.                      # linkes Lager auf x=0 setzen

numNodes        = n
FixedNodeIndices = [0, n - 1]

print("Knotenkoordinaten (x, y):")
for i in range(n):
    print(f"  Knoten {i:2d}: ({NodePos[0,i]:.1f}, {NodePos[1,i]:.1f})")


# --- Konnektivitätsmatrix ---
C = np.zeros((n, n))
for i in range(n - 3):
    if i % 2 == 0:
        C[i, [i + 1, i + 2]] = 1       # Untergurt + Diagonale nach oben rechts
    else:
        C[i, [i + 1, i + 2, i + 3]] = 1  # Diagonale + Untergurt + Obergurt
C[n - 3, [n - 2, n - 1]] = 1           # letztes Paar Richtung rechtes Lager
C[n - 2, n - 1] = 1
C = C + C.T

print(f"\nAnzahl Stäbe: {int(C.sum() / 2)}")


# --- Fachwerk visualisieren ---
def plotBruecke(UTruss=None, scalingFactor=5000, displayForces=False,
                FTruss_plot=None, title=''):
    if UTruss is None:
        UTruss = np.zeros(2 * numNodes)
    if FTruss_plot is None:
        FTruss_plot = np.zeros(2 * numNodes)

    fig, ax = plt.subplots(figsize=(10, 4))
    NodePos_d = NodePos + scalingFactor * UTruss.reshape((numNodes, 2)).T

    for i in range(numNodes):
        for j in range(i + 1, numNodes):
            if C[i, j]:
                ax.plot([NodePos[0,i],   NodePos[0,j]],
                        [NodePos[1,i],   NodePos[1,j]],
                        color='gray', linewidth=1, alpha=0.25)
                ax.plot([NodePos_d[0,i], NodePos_d[0,j]],
                        [NodePos_d[1,i], NodePos_d[1,j]],
                        color='tab:blue', linewidth=2)

    ax.scatter(NodePos[0,:],   NodePos[1,:],   c='gray',    s=40, zorder=4, alpha=0.3)
    ax.scatter(NodePos_d[0,:], NodePos_d[1,:], c='tab:red', s=60, zorder=5)
    for ni in range(numNodes):
        ax.text(NodePos_d[0,ni] + 0.04, NodePos_d[1,ni] + 0.04,
                str(ni), fontsize=7)

    hw = 0.08
    for ni in FixedNodeIndices:
        tx = [NodePos_d[0,ni], NodePos_d[0,ni]-hw/2, NodePos_d[0,ni]+hw/2]
        ty = [NodePos_d[1,ni], NodePos_d[1,ni]-hw,   NodePos_d[1,ni]-hw]
        ax.fill(tx, ty, color='tab:green', alpha=0.7)

    if displayForces:
        for ni in range(numNodes):
            if abs(FTruss_plot[2*ni+1]) > 0.01:
                ax.annotate(f'{FTruss_plot[2*ni+1]:.0f} N',
                            xy=(NodePos_d[0,ni], NodePos_d[1,ni]),
                            xytext=(NodePos_d[0,ni], NodePos_d[1,ni] - 0.25),
                            fontsize=6, ha='center',
                            arrowprops=dict(arrowstyle='->', color='tab:orange'))

    ax.set_title(title)
    ax.set_aspect('equal')
    ax.grid(True)
    plt.tight_layout()
    plt.show()


plotBruecke(title='Fachwerkbrücke in Ausgangslage')
```

## Eigengewicht der Stäbe als Knotenlast

Jeder Stab hat ein Eigengewicht, das auf seine beiden Endknoten verteilt
wird. Wir nehmen vereinfachend an, dass jede Hälfte des Stabgewichts als
vertikale Kraft nach unten auf den jeweiligen Endknoten wirkt:

$$F_{\text{Eigengewicht, Knoten } n} = -\frac{1}{2} \sum_{s \in \text{Stäbe bei } n} \rho \cdot A \cdot l_s \cdot g$$

Dabei ist $l_s$ die Länge des Stabs $s$. Das Minuszeichen gibt die
Richtung nach unten an.

```{code-cell} python
# --- Eigengewicht ---
# Knotenkraftvektor: zunächst alles null
extForcesOnNodes = np.zeros((numNodes, 2))

# Für jeden Stab: Gewicht berechnen und je die Hälfte auf beide Endknoten
for i in range(numNodes):
    for j in range(i + 1, numNodes):
        if C[i, j]:
            laenge   = np.linalg.norm(NodePos[:, j] - NodePos[:, i])
            stabgewicht = rho * A * laenge * g     # Gewichtskraft des Stabs in N

            # Halbes Stabgewicht auf jeden Endknoten (nach unten = negativ)
            extForcesOnNodes[i, 1] -= stabgewicht / 2.
            extForcesOnNodes[j, 1] -= stabgewicht / 2.

FTruss = extForcesOnNodes.flatten()

# Eigengewicht ausgeben (nur Knoten mit Last)
print("Eigengewichtslasten an den Knoten:")
for i in range(numNodes):
    fy = extForcesOnNodes[i, 1]
    if abs(fy) > 1e-6:
        print(f"  Knoten {i:2d}: Fy = {fy:.4f} N")

print(f"\nGesamtgewicht (Summe aller Lasten): {FTruss[1::2].sum():.4f} N")
```

## Lösung und Auswertung

Steifigkeitsmatrix, Reduktion und Lösung laufen identisch zu den
vorherigen Kapiteln ab. Wir lösen jetzt das vollständige System mit
Eigengewicht.

```{code-cell} python
# --- Globale Steifigkeitsmatrix ---
KTruss = np.zeros((2 * numNodes, 2 * numNodes))
for i in range(numNodes):
    for j in range(i + 1, numNodes):
        if C[i, j]:
            diff   = NodePos[:, j] - NodePos[:, i]
            laenge = np.linalg.norm(diff)
            phi    = np.arctan2(diff[1], diff[0])
            k      = E * A / laenge
            cp, sp = np.cos(phi), np.sin(phi)
            Kij    = k * np.array([[cp**2, sp*cp],[sp*cp, sp**2]])
            KTruss[2*i:2*(i+1), 2*i:2*(i+1)] += Kij
            KTruss[2*j:2*(j+1), 2*j:2*(j+1)] += Kij
            KTruss[2*i:2*(i+1), 2*j:2*(j+1)] -= Kij
            KTruss[2*j:2*(j+1), 2*i:2*(i+1)] -= Kij

# --- Reduzieren und Lösen ---
FreeNodeIndices = [ni for ni in range(numNodes) if ni not in FixedNodeIndices]
reduceMask = []
for ni in FreeNodeIndices:
    reduceMask += [2*ni, 2*ni + 1]
reduceMask = np.array(reduceMask)

Ured   = np.linalg.solve(KTruss[reduceMask, :][:, reduceMask], FTruss[reduceMask])
UTruss = np.zeros(2 * numNodes)
UTruss[reduceMask] = Ured

FTruss_result = np.round(KTruss @ UTruss, 8)

# --- Maximale Absenkung ---
uy_alle = UTruss[1::2]   # alle y-Verschiebungen
idx_max_absenkung = np.argmin(uy_alle)
print(f"Maximale Absenkung: Knoten {idx_max_absenkung}, "
      f"uy = {uy_alle[idx_max_absenkung]*1e3:.4f} mm")

# --- Verformte Brücke ---
plotBruecke(UTruss=UTruss, scalingFactor=5000,
            displayForces=True, FTruss_plot=FTruss_result,
            title='Verformte Brücke unter Eigengewicht (5000-fach überhöht)')
```

uy_nutz = UTruss_nutz[1::2]

steigerung = (absenkung_nutz - absenkung_eigen) / absenkung_eigen * 100.
print(f"Absenkung Eigengewicht:        {absenkung_eigen*1e3:.4f} mm")
print(f"Absenkung Eigengewicht + Nutz: {absenkung_nutz*1e3:.4f} mm")
print(f"Steigerung:                    {steigerung:.1f} %")

# Frage 3: Doppelter Durchmesser -> A_neu = 4*A
# k_neu = E*A_neu/l = 4*k   -> Steifigkeit 4x größer  (Nenner wächst)
# F_gew_neu = rho*A_neu*l*g = 4*F_gew  (Zähler wächst um denselben Faktor)
# Absenkung ~ F/k = (4*F)/(4*k) = F/k -> unverändert!
# Eigengewichtsabsenkung ist unabhängig vom Querschnitt (bei gleichem Material).
```

Bei reinem Eigengewicht ist die Absenkung unabhängig vom Querschnitt:
$u \propto F/k \propto (\rho A l g) / (EA/l) = \rho l^2 g / E$. Querschnitt
$A$ kürzt sich heraus. Nur bei einer externen Nutzlast hilft ein größerer
Querschnitt, weil dann nur der Nenner $k \propto A$ wächst.
````

## Zusammenfassung und Ausblick

Das Eigengewicht der Stäbe wird als vertikale Knotenlast berücksichtigt:
jeder Stab übergibt die Hälfte seines Gewichts $\rho A l g$ an jeden seiner
Endknoten. Damit fügt sich Eigengewicht nahtlos in den bestehenden
Kraftvektor ein, ohne den FEM-Algorithmus zu ändern.
